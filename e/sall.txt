نظام درجات الطلبة
الموضوع: المصفوفات أحادية البعد والدوال.
اكتب برنامجًا بلغة C++ لإدارة درجات خمسة طلبة في إحدى المواد الدراسية، وفق المتطلبات الآتية:
استخدام مصفوفة لتخزين درجات الطلبة الخمسة.
إنشاء دالة باسم inputGrades() لإدخال الدرجات.
إنشاء دالة باسم calculateAverage() لحساب المعدل العام للدرجات وإرجاعه.
إنشاء دالة باسم findHighest() لإيجاد أعلى درجة.
طباعة المعدل العام وأعلى درجة.



#include <iostream>
using namespace std;

void inputGrades(int grades[], int size) {
    for (int i = 0; i < size; i++) {
        cout << "Enter grade " << i + 1 << ": ";
        cin >> grades[i];
    }
}

double calculateAverage(int grades[], int size) {
    int sum = 0;

    for (int i = 0; i < size; i++) {
        sum += grades[i];
    }

    return (double)sum / size;
}

int findHighest(int grades[], int size) {
    int highest = grades[0];

    for (int i = 1; i < size; i++) {
        if (grades[i] > highest) {
            highest = grades[i];
        }
    }

    return highest;
}

int main() {
    int grades[5];

    inputGrades(grades, 5);

    cout << "Average = " << calculateAverage(grades, 5) << endl;
    cout << "Highest Grade = " << findHighest(grades, 5) << endl;

    return 0;
}

نظام المبيعات في متجر
الموضوع: المصفوفات والدوال التي تستقبل معاملات.
يمتلك متجر خمسة منتجات، ولكل منتج كمية مباعة وسعر محدد. اكتب برنامجًا بلغة C++ ينفذ الآتي:
تخزين أسعار المنتجات في مصفوفة.
تخزين الكميات المباعة في مصفوفة أخرى.
كتابة دالة باسم calculateSales() تستقبل مصفوفتَي الأسعار والكميات وتحسب إجمالي المبيعات لكل المنتجات.
كتابة دالة باسم findBestSeller() لتحديد المنتج الذي حقق أعلى قيمة مبيعات.
عرض إجمالي المبيعات واسم أو رقم المنتج الأعلى مبيعًا.




#include <iostream>
using namespace std;

double calculateSales(double prices[], int quantities[], int size) {
    double total = 0;

    for (int i = 0; i < size; i++) {
        total += prices[i] * quantities[i];
    }

    return total;
}

int findBestSeller(double prices[], int quantities[], int size) {
    int best = 0;

    for (int i = 1; i < size; i++) {
        if (prices[i] * quantities[i] > prices[best] * quantities[best]) {
            best = i;
        }
    }

    return best;
}

int main() {
    double prices[5] = {100, 200, 150, 300, 250};
    int quantities[5] = {10, 5, 8, 12, 6};

    cout << "Total Sales = "
         << calculateSales(prices, quantities, 5) << endl;

    int best = findBestSeller(prices, quantities, 5);

    cout << "Best Selling Product = " << best + 1 << endl;

    return 0;
}

يريد مركز الأرصاد الجوية تسجيل درجات الحرارة خلال سبعة أيام. اكتب برنامجًا بلغة C++ يحقق المتطلبات الآتية:
تخزين درجات الحرارة اليومية في مصفوفة.
إنشاء دالة باسم calculateAverage() لحساب متوسط درجات الحرارة.
إنشاء دالة باسم countHotDays() لحساب عدد الأيام التي تجاوزت فيها درجة الحرارة درجة مئوية.
إنشاء دالة باسم findLowest() لإيجاد أقل درجة حرارة مسجلة.
طباعة النتائج بصورة منظمة.




#include <iostream>
using namespace std;

double calculateAverage(int temp[], int size) {
    int sum = 0;

    for (int i = 0; i < size; i++) {
        sum += temp[i];
    }

    return (double)sum / size;
}

int countHotDays(int temp[], int size) {
    int count = 0;

    for (int i = 0; i < size; i++) {
        if (temp[i] > 30) {
            count++;
        }
    }

    return count;
}

int findLowest(int temp[], int size) {
    int lowest = temp[0];

    for (int i = 1; i < size; i++) {
        if (temp[i] < lowest) {
            lowest = temp[i];
        }
    }

    return lowest;
}

int main() {
    int temp[7] = {25, 32, 28, 35, 30, 33, 22};

    cout << "Average Temperature = "
         << calculateAverage(temp, 7) << endl;

    cout << "Hot Days = "
         << countHotDays(temp, 7) << endl;

    cout << "Lowest Temperature = "
         << findLowest(temp, 7) << endl;

    return 0;
}

نظام إدارة مخزون مختبر الحاسوب
الموضوع: المصفوفات والدوال متعددة المهام.
يحتوي مختبر الحاسوب على خمسة أنواع من الأجهزة، ويريد مسؤول المختبر متابعة أعدادها. اكتب برنامجًا بلغة C++ ينفذ ما يأتي:
تخزين أسماء الأجهزة وأعدادها في مصفوفتين.
إنشاء دالة باسم displayInventory() لعرض جميع الأجهزة وأعدادها.
إنشاء دالة باسم searchDevice() للبحث عن جهاز محدد باستخدام رقمه في المصفوفة.
إنشاء دالة باسم updateQuantity() لتحديث عدد جهاز بعد إضافة أجهزة جديدة أو إخراج أجهزة من الخدمة.
عرض المخزون بعد إجراء التحديث.




#include <iostream>
#include <string>
using namespace std;

void displayInventory(string devices[], int quantities[], int size) {
    cout << "\n--- Computer Lab Inventory ---\n";

    for (int i = 0; i < size; i++) {
        cout << i + 1 << ". " << devices[i]
             << " - Quantity: " << quantities[i] << endl;
    }
}

int searchDevice(string devices[], int quantities[], int size, int number) {
    if (number >= 1 && number <= size) {
        cout << "Device: " << devices[number - 1] << endl;
        cout << "Quantity: " << quantities[number - 1] << endl;
        return number - 1;
    }

    cout << "Device Not Found" << endl;
    return -1;
}

void updateQuantity(int quantities[], int size, int index, int change) {
    if (index >= 0 && index < size) {
        if (quantities[index] + change >= 0) {
            quantities[index] += change;
            cout << "Quantity Updated Successfully" << endl;
        } else {
            cout << "Error: Quantity cannot be negative" << endl;
        }
    }
}

int main() {
    string devices[5] = {
        "Computer", "Monitor", "Keyboard", "Mouse", "Printer"
    };

    int quantities[5] = {10, 8, 12, 15, 3};
    int number, change;

    displayInventory(devices, quantities, 5);

    cout << "\nEnter device number to search (1-5): ";
    cin >> number;

    int index = searchDevice(devices, quantities, 5, number);

    if (index != -1) {
        cout << "\nEnter quantity change (+ to add, - to remove): ";
        cin >> change;

        updateQuantity(quantities, 5, index, change);

        displayInventory(devices, quantities, 5);
    }

    return 0;
}

نظام حساب رواتب الموظفين
الموضوع: المصفوفات والدوال مع المعاملات والقيم المرجعة.
تريد إحدى الشركات حساب الرواتب الشهرية لخمسة موظفين. اكتب برنامجًا بلغة C++ وفق المتطلبات الآتية:
تخزين الرواتب الأساسية للموظفين في مصفوفة.
إنشاء دالة باسم calculateSalary() تستقبل الراتب الأساسي ومقدار المكافأة، ثم تعيد الراتب النهائي.
إنشاء دالة باسم calculateTotalPayroll() لحساب مجموع الرواتب النهائية لجميع الموظفين.
إنشاء دالة باسم findHighestSalary() لإيجاد أعلى راتب.
عرض راتب كل موظف ومجموع الرواتب وأعلى راتب.
ملاحظة: الراتب النهائي = الراتب الأساسي + المكافأة.



#include <iostream>
using namespace std;

double calculateSalary(double basicSalary, double bonus) {
    return basicSalary + bonus;
}

double calculateTotalPayroll(double salaries[], double bonus, int size) {
    double total = 0;

    for (int i = 0; i < size; i++) {
        total += calculateSalary(salaries[i], bonus);
    }

    return total;
}

double findHighestSalary(double salaries[], double bonus, int size) {
    double highest = calculateSalary(salaries[0], bonus);

    for (int i = 1; i < size; i++) {
        double salary = calculateSalary(salaries[i], bonus);

        if (salary > highest) {
            highest = salary;
        }
    }

    return highest;
}

int main() {
    double salaries[5] = {500000, 600000, 450000, 700000, 550000};
    double bonus = 50000;

    for (int i = 0; i < 5; i++) {
        cout << "Employee " << i + 1 << " Salary = "
             << calculateSalary(salaries[i], bonus) << endl;
    }

    cout << "Total Payroll = "
         << calculateTotalPayroll(salaries, bonus, 5) << endl;

    cout << "Highest Salary = "
         << findHighestSalary(salaries, bonus, 5) << endl;

    return 0;
}

نظام تسجيل الدخول والتحقق من أرقام PIN
الموضوع: المصفوفات، الدوال، البحث، والتحقق من البيانات.
تريد مؤسسة إنشاء برنامج مبسط للتحقق من أرقام PIN الخاصة بخمسة حسابات تجريبية. اكتب برنامجًا بلغة C++ ينفذ الآتي:
تخزين أرقام PIN التجريبية في مصفوفة من الأعداد الصحيحة.
إنشاء دالة باسم verifyPIN() تستقبل المصفوفة ورقم PIN المدخل، وتعيد true إذا كان الرقم مطابقًا لأحد الأرقام المسجلة، وإلا تعيد false.
السماح للمستخدم بإدخال رقم PIN للتحقق منه.
عرض رسالة توضح نجاح التحقق أو فشله.
السماح بثلاث محاولات فقط، ثم إيقاف عملية التحقق.





#include <iostream>
using namespace std;

bool verifyPIN(int pins[], int size, int enteredPIN) {
    for (int i = 0; i < size; i++) {
        if (pins[i] == enteredPIN) {
            return true;
        }
    }

    return false;
}

int main() {
    int pins[5] = {1234, 5678, 4321, 9876, 2468};
    int enteredPIN;
    bool success = false;

    for (int i = 0; i < 3; i++) {
        cout << "Enter PIN: ";
        cin >> enteredPIN;

        if (verifyPIN(pins, 5, enteredPIN)) {
            cout << "Login Successful" << endl;
            success = true;
            break;
        } else {
            cout << "Incorrect PIN" << endl;
        }
    }

    if (!success) {
        cout << "Verification Failed. No attempts left." << endl;
    }

    return 0;
}
